static inline float wrap180(float a) {
  while (a >  180.f) a -= 360.f;
  while (a < -180.f) a += 360.f;
  return a;
}

// Slew toward target PWM, honoring ESC range
static uint16_t slewThrottleToward(uint16_t targetUs) {
  uint32_t now = micros();
  float dt = (g_lastSlewMicros == 0) ? 0.005f : (now - g_lastSlewMicros) * 1e-6f;
  g_lastSlewMicros = now;
  dt = fminf(fmaxf(dt, 0.0005f), 0.02f);

  float step = THROTTLE_SLEW_US_PER_S * dt;
  if (g_slewThrUs < targetUs) g_slewThrUs = fminf(g_slewThrUs + step, (float)targetUs);
  else                        g_slewThrUs = fmaxf(g_slewThrUs - step, (float)targetUs);

  g_slewThrUs = fminf(fmaxf(g_slewThrUs, (float)THROTTLE_MIN_US), (float)THROTTLE_MAX_US);
  return (uint16_t)g_slewThrUs;
}

// ====== Robust I2C helpers ======
const int SDA_PIN_REC = SDA_PIN;
const int SCL_PIN_REC = SCL_PIN;
uint32_t lastIMUokMs = 0;

bool i2cBusRecover() {
  pinMode(SDA_PIN_REC, INPUT_PULLUP);
  pinMode(SCL_PIN_REC, INPUT_PULLUP);
  delay(1);
  for (int i = 0; i < 9 && digitalRead(SDA_PIN_REC) == LOW; ++i) {
    pinMode(SCL_PIN_REC, OUTPUT);
    digitalWrite(SCL_PIN_REC, LOW);
    delayMicroseconds(5);
    pinMode(SCL_PIN_REC, INPUT_PULLUP);
    delayMicroseconds(5);
  }
  pinMode(SDA_PIN_REC, OUTPUT);
  digitalWrite(SDA_PIN_REC, LOW);
  delayMicroseconds(5);
  pinMode(SCL_PIN_REC, INPUT_PULLUP);
  delayMicroseconds(5);
  pinMode(SDA_PIN_REC, INPUT_PULLUP);
  delayMicroseconds(5);

  Wire.begin(SDA_PIN, SCL_PIN, I2C_HZ);
  Wire.setTimeOut(50);
  return (digitalRead(SDA_PIN_REC) == HIGH) && (digitalRead(SCL_PIN_REC) == HIGH);
}

bool readIMU_safe() {
  for (int t = 0; t < 3; ++t) {
    if (!imu.dataReady()) { delay(2); continue; }
    (void)imu.getAGMT();           // fills imu.agmt and updates imu.status
    if (imu.status == ICM_20948_Stat_Ok) {
      lastIMUokMs = millis();
      return true;
    }
    delay(5);
  }
  return false;
}

void recoverIMU() {
  i2cBusRecover();             // free bus if stuck
  imu.swReset();               // soft reset the IMU
  delay(10);
  imu.begin(Wire, AD0_VAL);    // re-begin
  delay(5);
}

void imuWatchdog(uint32_t staleMs = 500) {
  if (millis() - lastIMUokMs > staleMs) {
    recoverIMU();
  }
}

// Throttle PCA9685 write rate to avoid clumping 4 writes together
bool allowServoWrite() {
  static uint32_t last = 0;
  uint32_t now = millis();
  if (now - last >= 3) { last = now; return true; } // 3–5 ms spacing helps
  return false;
}

// ==== IMU init & calibration ====
bool initIMU() {
  Wire.begin(SDA_PIN, SCL_PIN, I2C_HZ);
  Wire.setTimeOut(50);                 // helps avoid 263-style timeouts
  ICM_20948_Status_e status = ICM_20948_Stat_Err;
  for (int i = 0; i < 10; i++) {
    status = imu.begin(Wire, AD0_VAL);
    if (status == ICM_20948_Stat_Ok) return true;
    delay(100);
  }
  return false;
}

void calibrateStill() {
  LOGILN("\n[Calib] Keep IMU still...");
  delay(500);

  double axSum=0, aySum=0, azSum=0;
  double gxSum=0, gySum=0, gzSum=0;
  uint16_t n = 0;

  while (n < CALIB_SAMPLES) {
    if (imu.dataReady()) {
      imu.getAGMT();

      float ax = to_g(imu.accX());
      float ay = to_g(imu.accY());
      float az = to_g(imu.accZ());

      float gx = imu.gyrX();
      float gy = imu.gyrY();
      float gz = imu.gyrZ();

      axSum += ax; aySum += ay; azSum += az;
      gxSum += gx; gySum += gy; gzSum += gz;
      n++;
    }
    delay(CALIB_DELAY_MS);
  }

  float axm = axSum / n, aym = aySum / n, azm = azSum / n;
  float gxm = gxSum / n, gym = gySum / n, gzm = gzSum / n;

  gyroBias[0] = gxm;
  gyroBias[1] = gym;
  gyroBias[2] = gzm;

  float normA = sqrtf(axm*axm + aym*aym + azm*azm);
  if (normA < 1e-6f) normA = 1.0f;

  float gxdir = axm / normA;
  float gydir = aym / normA;
  float gzdir = azm / normA;

  accelBias[0] = axm - gxdir;
  accelBias[1] = aym - gydir;
  accelBias[2] = azm - gzdir;

  LOGILN("[Calib] Complete.");
}

// === Helpers ===
static inline float clampf(float x, float a, float b){ return x < a ? a : (x > b ? b : x); }

// Convert quaternion to R/P/Y (deg)
static inline void quatToRPYdeg(float w,float x,float y,float z,
                                float &roll,float &pitch,float &yaw){
  float siny = 2.0f * (w*y - z*x);
  pitch = asinf(clampf(siny, -1.0f, 1.0f)) * 180.0f / PI;

  float sinr_cosp = 2.0f * (w*x + y*z);
  float cosr_cosp = 1.0f - 2.0f * (x*x + y*y);
  roll = atan2f(sinr_cosp, cosr_cosp) * 180.0f / PI;

  float siny_cosp = 2.0f * (w*z + x*y);
  float cosy_cosp = 1.0f - 2.0f * (y*y + z*z);
  yaw  = atan2f(siny_cosp, cosy_cosp) * 180.0f / PI;
}

// Mahony 6-axis update (no mag). Units: accel in g, gyro in rad/s, dt in s
void mahony6Update(float ax,float ay,float az, float gx,float gy,float gz, float dt){
  float norm = sqrtf(ax*ax + ay*ay + az*az);
  if (norm < 1e-6f) return;
  ax/=norm; ay/=norm; az/=norm;

  float vx = 2.0f*(q1*q3 - q0*q2);
  float vy = 2.0f*(q0*q1 + q2*q3);
  float vz = q0*q0 - q1*q1 - q2*q2 + q3*q3;

  float ex = (ay*vz - az*vy);
  float ey = (az*vx - ax*vz);
  float ez = (ax*vy - ay*vx);

  if (twoKi > 0.0f){
    ix += twoKi * ex * dt;
    iy += twoKi * ey * dt;
    iz += twoKi * ez * dt;
  } else {
    ix = iy = iz = 0.0f;
  }

  gx += twoKp * ex + ix;
  gy += twoKp * ey + iy;
  gz += twoKp * ez + iz;

  float qw = q0, qx = q1, qy = q2, qz = q3;
  float halfDt = 0.5f * dt;
  q0 += (-qx*gx - qy*gy - qz*gz) * halfDt;
  q1 += ( qw*gx + qy*gz - qz*gy) * halfDt;
  q2 += ( qw*gy - qx*gz + qz*gx) * halfDt;
  q3 += ( qw*gz + qx*gy - qy*gx) * halfDt;

  float n = sqrtf(q0*q0 + q1*q1 + q2*q2 + q3*q3);
  if (n > 0.0f){ q0/=n; q1/=n; q2/=n; q3/=n; }
}

// Online gyro-bias update (when device is clearly still)
void updateGyroBiasOnline(float ax_g, float ay_g, float az_g,
                          float gx_raw, float gy_raw, float gz_raw,
                          float dt)
{
  float amag = sqrtf(ax_g*ax_g + ay_g*ay_g + az_g*az_g);
  float alpha = 1.0f - expf(-2.0f * PI * 2.0f * dt);  // ~2 Hz LP
  float err = fabsf(amag - aMagLP);
  aMagLP += alpha * (amag - aMagLP);
  aDevLP += alpha * (err   - aDevLP);

  bool gyroQuiet = (fabsf(gx_raw) < GYRO_STILL_DPS &&
                    fabsf(gy_raw) < GYRO_STILL_DPS &&
                    fabsf(gz_raw) < GYRO_STILL_DPS);
  bool accelNear1g = (fabsf(amag - 1.0f) < AMAG_TOL_G);
  bool accelQuiet  = (aDevLP < ADEV_TOL_G);

  bool still = gyroQuiet && accelNear1g && accelQuiet;
  if (!still) return;

  float a = dt / BIAS_TAU_S;
  gyroBias[0] = (1.0f - a) * gyroBias[0] + a * gx_raw;
  gyroBias[1] = (1.0f - a) * gyroBias[1] + a * gy_raw;
  gyroBias[2] = (1.0f - a) * gyroBias[2] + a * gz_raw;
}

// ===== OTA task (core 0), high priority, AP tuned =====
TaskHandle_t otaTaskHandle = nullptr;
volatile bool g_otaActive = false;

void otaTask(void*) {
  WiFi.mode(WIFI_AP);
  WiFi.setSleep(false);

  WiFi.softAPConfig(IPAddress(192,168,4,1),
                    IPAddress(192,168,4,1),
                    IPAddress(255,255,255,0));
  WiFi.softAP(AP_SSID, AP_PASS, 1 /*ch*/, 0 /*visible*/, 1 /*clients*/);

  // IDF APIs for compatibility
  esp_wifi_set_bandwidth(WIFI_IF_AP, WIFI_BW_HT20);
  esp_wifi_set_max_tx_power(78);  // 19.5 dBm

  IPAddress apIP = WiFi.softAPIP();
  LOGI("[OTA] AP up. SSID: %s  IP: %s\n", AP_SSID, apIP.toString().c_str());

  ArduinoOTA.setHostname(OTA_HOSTNAME);
  ArduinoOTA.setPassword(AP_PASS);
  ArduinoOTA.setPort(OTA_PORT);

  ArduinoOTA.onStart([](){
    g_otaActive = true;
    esc.writeMicroseconds(1000);
    servo.resetServos(90);
    ps5.end();
    LOGILN("OTA start");
  });
  ArduinoOTA.onEnd([](){
    LOGILN("OTA end");
    delay(200);
    ps5.begin(PS5_MAC);
    g_otaActive = false;
  });
  ArduinoOTA.onError([](ota_error_t e){
    LOGI("OTA error: %d\n", e);
    delay(200);
    ps5.begin(PS5_MAC);
    g_otaActive = false;
  });

  ArduinoOTA.begin();

  for (;;) {
    ArduinoOTA.handle();
    vTaskDelay(pdMS_TO_TICKS(2));
  }
}

// ===== setup / loop =====
void setup() {
  Serial.begin(SERIAL_BAUD);

  // Start OTA FIRST (lifeline), higher priority on core 0
  xTaskCreatePinnedToCore(
    otaTask, "otaTask",
    4096,
    nullptr,
    4,           // higher prio favors OTA radio/CPU time
    &otaTaskHandle,
    0            // run on core 0
  );

  // Maintenance window: give yourself ~5 s where only OTA runs
  uint32_t tStart = millis();
  while (millis() - tStart < 5000) {
    if (g_otaActive) break;
    delay(50);
  }

  if (!g_otaActive) {
    if (!initIMU()) {
      LOGILN("IMU init failed. Check wiring/address.");
      while (true) delay(1000);   // OTA still available
    }
    calibrateStill();

    t0_ms = millis();
    headerPrinted = false;
    yawZeroed = false;

    servo.begin();

    // PCA I2C setup (same shared bus, slow it too)
    Servo_Module::configureBus(0x40, SDA_PIN, SCL_PIN, I2C_HZ, 50);

    ps5.begin(PS5_MAC);

    esc.writeMicroseconds(1000);
    servo.resetServos(90);
    servo.initServoAngles();
    delay(launchDelay);
    missionActive = true;
    mstate = MS_ARMED;
  }
}

// NOTE: signature now includes rollDeg so we can capture start roll cleanly at ARM.
static void updateMission(float dt, float pitchDeg, float yawDeg, float rollDeg)
{
  if (!missionActive) return;

  // Safety conditions → abort
  if (g_otaActive)          { mstate = MS_ABORT; }
  if (killSwitchEngaged)    { mstate = MS_ABORT; }

  switch (mstate)
  {
    case MS_ARMED:
    {
      // Capture refs at arm
      yawRefDeg     = yawDeg;          // hold current heading
      pitchRefDeg   = pitchDeg;        // hold current pitch (not necessarily 0)

      // NEW: capture roll start exactly at ARM (for ±60° window)
      g_rollStartDeg   = rollDeg;
      g_rollStartValid = true;
      g_rollFault      = false;
      g_allStop        = false;
      g_faultT0 = g_recoverT0 = 0;

      autoThrottle = IDLE_THROTTLE_US;

      // Init timers/accums
      phaseT0ms = millis();
      legTimeS  = 0.0f;
      turnHoldS = 0.0f;

      // Apply initial targets (clamped)
      float pt = clampf_deg(pitchRefDeg, -SERVO_MAX_ABS_DEG, SERVO_MAX_ABS_DEG);
      servo.setTargetAngles(pt, yawRefDeg);

      mstate = MS_DIVE;
      break;
    }

    case MS_DIVE:
    {
      // Fixed-time dive with small nose-down bias
      autoThrottle = DIVE_THROTTLE_US;

      float pt = clampf_deg(pitchRefDeg + DIVE_PITCH_BIAS, -SERVO_MAX_ABS_DEG, SERVO_MAX_ABS_DEG);
      servo.setTargetAngles(pt, yawRefDeg);

      if (millis() - phaseT0ms >= DIVE_TIME_MS) {
        // Move to outbound cruise
        phaseT0ms = millis();
        legTimeS  = 0.0f;
        autoThrottle = CRUISE_THROTTLE_US;

        float ppt = clampf_deg(pitchRefDeg, -SERVO_MAX_ABS_DEG, SERVO_MAX_ABS_DEG);
        servo.setTargetAngles(ppt, yawRefDeg);
        mstate = MS_OUTBOUND;
      }
      break;
    }

    case MS_OUTBOUND:
    {
      autoThrottle = CRUISE_THROTTLE_US;

      float pt = clampf_deg(pitchRefDeg, -SERVO_MAX_ABS_DEG, SERVO_MAX_ABS_DEG);
      servo.setTargetAngles(pt, yawRefDeg);

      legTimeS += dt;
      if (legTimeS >= LEG_TIME_S) {
        // Prepare the 180° turn
        yawRefDeg = norm180(yawRefDeg + 170.0f);
        turnHoldS = 0.0f;
        turnT0ms  = millis();
        mstate = MS_TURN;
      }
      break;
    }

    case MS_TURN:
    {
      autoThrottle = TURN_THROTTLE_US;

      float pt = clampf_deg(pitchRefDeg, -SERVO_MAX_ABS_DEG, SERVO_MAX_ABS_DEG);
      servo.setTargetAngles(pt, yawRefDeg);

      float e = fabsf(yawErrDeg(yawRefDeg, yawDeg));
      if (e <= (float)YAW_TOL_DEG) {
        turnHoldS += dt;
      } else {
        turnHoldS = 0.0f;
      }

      if (turnHoldS >= (TURN_HOLD_MS / 1000.0f)) {
        // Turn complete → inbound
        phaseT0ms = millis();
        legTimeS  = 0.0f;
        mstate = MS_INBOUND;
      } else if (millis() - turnT0ms > TURN_TIMEOUT_MS) {
        // Timeout → accept-best and proceed
        phaseT0ms = millis();
        legTimeS  = 0.0f;
        mstate = MS_INBOUND;
      }
      break;
    }

    case MS_INBOUND:
    {
      autoThrottle = CRUISE_THROTTLE_US;

      float pt = clampf_deg(pitchRefDeg, -SERVO_MAX_ABS_DEG, SERVO_MAX_ABS_DEG);
      servo.setTargetAngles(pt, yawRefDeg);

      legTimeS += dt;
      if (legTimeS >= LEG_TIME_S) {
        mstate = MS_DONE;
      }
      break;
    }

    case MS_DONE:
    default:
    {
      // Idle + neutral fins (or keep pitchRef; pick A or B)
      autoThrottle = IDLE_THROTTLE_US;

      // A) neutralize fins:
      servo.setTargetAngles(pitchRefDeg, yawRefDeg);
      // B) keep pitchRef (comment A, uncomment B):
      // float pt = clampf_deg(pitchRefDeg, -SERVO_MAX_ABS_DEG, SERVO_MAX_ABS_DEG);
      // servo.setTargetAngles(pt, yawDeg);

      missionActive = false;
      mstate = MS_IDLE;
      break;
    }

    case MS_ABORT:
    {
      autoThrottle = IDLE_THROTTLE_US;
      servo.setTargetAngles(0.0f, 0.0f);
      missionActive = false;
      mstate = MS_IDLE;
      break;
    }
  }
}

void loop() {
  // If OTA is in progress, pause everything to keep radio happy
  if (g_otaActive) { delay(10); return; }

  // ===== PS5 handling =====
  static bool wasConnected = false;

  // ===== IMU processing =====
  if (!readIMU_safe()) {
    imuWatchdog(500);
    delay(1);
    return;
  }

  // RAW readings (before bias)
  float ax_g_raw = to_g(imu.accX());
  float ay_g_raw = to_g(imu.accY());
  float az_g_raw = to_g(imu.accZ());
  float gx_raw   = imu.gyrX();   // dps
  float gy_raw   = imu.gyrY();
  float gz_raw   = imu.gyrZ();

  // dt (s) with clamp
  unsigned long now = micros();
  float dt = (lastMicros==0) ? 0.005f : (now - lastMicros) * 1e-6f;
  lastMicros = now;
  dt = fminf(fmaxf(dt, 0.0005f), 0.02f);  // 0.5–20 ms

  // Online bias adaptation (only when clearly still)
  updateGyroBiasOnline(ax_g_raw, ay_g_raw, az_g_raw, gx_raw, gy_raw, gz_raw, dt);

  // Bias-corrected readings
  float ax = ax_g_raw - accelBias[0];
  float ay = ay_g_raw - accelBias[1];
  float az = az_g_raw - accelBias[2];
  float gx = gx_raw   - gyroBias[0];
  float gy = gy_raw   - gyroBias[1];
  float gz = gz_raw   - gyroBias[2];

  // Moving average on gz for smoother yaw integration
  gzMA -= gzBuf[gzIdx];
  gzBuf[gzIdx] = gz;
  gzMA += gz;
  gzIdx = (gzIdx + 1) % GZ_WIN;
  if (gzFilled < GZ_WIN) gzFilled++;
  float gzFiltered = gzMA / (float)gzFilled;

  // Mahony update
  const float D2R = PI/180.0f;
  mahony6Update(ax, ay, az, gx*D2R, gy*D2R, gzFiltered*D2R, dt);

  // Roll / Pitch / Yaw (yaw zeroed only for verbose logging)
  float rollDeg, pitchDeg, yawDeg;
  quatToRPYdeg(q0, q1, q2, q3, rollDeg, pitchDeg, yawDeg);

  if (!yawZeroed) { yaw0 = yawDeg; yawZeroed = true; }
  float yawRel = yawDeg - yaw0;
  if (yawRel > 180.0f)  yawRel -= 360.0f;
  if (yawRel < -180.0f) yawRel += 360.0f;

  // Run mission logic (now also passes rollDeg so ARM can capture it)
  updateMission(dt, pitchDeg, yawDeg, rollDeg);

  Serial.print(yawDeg);
  Serial.print(" ");
  Serial.print(yawRefDeg);
  Serial.print(" ");

  // Roll safety supervisor
  // updateRollSafety(rollDeg);   // <— ACTIVATED

  // Throttle ownership: mission only. Otherwise keep idle.
  // Apply roll-safety and slew-limited output.
  uint16_t thrCmd = IDLE_THROTTLE_US;
  if (g_allStop)            thrCmd = IDLE_THROTTLE_US;
  else if (g_rollFault)     thrCmd = ROLL_FAULT_THROTTLE_US;
  else if (missionActive)   thrCmd = autoThrottle;
  else                      thrCmd = IDLE_THROTTLE_US;

  uint16_t thrOut = slewThrottleToward(thrCmd);
  esc.writeMicroseconds(thrOut);

  // Fin controller runs every cycle (unless OTA / kill)
  if (!g_otaActive && !killSwitchEngaged) {
    if (allowServoWrite()) {
      if (g_rollFault) {
        servo.setTargetAngles(0.0f, 0.0f);   // hold neutral during fault
      } else {
        servo.update(pitchDeg, yawDeg);
      }
    }
  }

#if LOG_LEVEL >= 2
  // Optional CSV stream (verbose only)
  if (!headerPrinted) {
    Serial.println(F("time_s,roll_deg,pitch_deg,yaw_deg"));
    headerPrinted = true;
  }
  float t = (millis() - t0_ms) * 0.001f;
  Serial.print(t, 3);     Serial.print(',');
  Serial.print(rollDeg, 2);  Serial.print(',');
  Serial.print(pitchDeg, 2); Serial.print(',');
  Serial.println(yawRel, 2);
#endif

  // ~200 Hz
  delay(5);
}

// ──────────────────────────────────────────────
// ROLL SAFETY: keeps fins neutral & drops throttle when |Δroll| > 60°
// Captures reference at ARM via g_rollStartDeg.
// ──────────────────────────────────────────────
static void updateRollSafety(float rollDeg)
{
  if (!g_rollStartValid) return;

  float d = norm180(rollDeg - g_rollStartDeg);
  float ad = fabsf(d);
  uint32_t nowMs = millis();

  if (ad > ROLL_FAULT_DEG) {
    // Enter/hold fault
    if (!g_rollFault) {
      g_rollFault = true;
      g_recoverT0 = 0;
      g_faultT0   = nowMs;
      LOGILN("[SAFETY] Roll fault entered");
    }
    if (ROLL_FAULT_TIMEOUT_MS && (nowMs - g_faultT0 > ROLL_FAULT_TIMEOUT_MS)) {
      g_allStop = true;   // give up after long fault
    }
  } else {
    // Inside the main band; if we are in fault, require recovery hysteresis + hold
    if (g_rollFault) {
      if (ad < ROLL_RECOVER_DEG) {
        if (g_recoverT0 == 0) g_recoverT0 = nowMs;
        if (nowMs - g_recoverT0 >= ROLL_RECOVER_HOLD_MS) {
          g_rollFault = false;
          g_allStop   = false;
          g_faultT0 = g_recoverT0 = 0;
          LOGILN("[SAFETY] Roll fault cleared");
        }
      } else {
        g_recoverT0 = 0; // bounced outside recovery band, reset hold timer
      }
    }
  }
}
